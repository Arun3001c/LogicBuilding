def children_satisfied(greedy_of_children, size_of_cookies):

    greedy_of_children.sort()
    size_of_cookies.sort()

    i = 0      # Child pointer
    j = 0      # Cookie pointer
    satisfied = 0

    while i < len(greedy_of_children) and j < len(size_of_cookies):

        if size_of_cookies[j] >= greedy_of_children[i]:
            satisfied += 1
            i += 1
            j += 1
        else:
            j += 1

    return satisfied


print(children_satisfied([2, 3, 5], [1, 2, 4, 6]))
