class Solution(object):
    def majorityElement(self, nums):
        count1=0
        count2=0
        can1=None
        can2=None
        res=[]
        n=len(nums)
        for num in nums:
            if can1==num:
                count1+=1
            elif can2==num:
                count2+=1
            elif count1==0:
                can1=num
                count1=1
            elif count2==0:
                can2=num
                count2=1
            else:
                count1-=1
                count2-=1
        for can in [can1,can2]:
            if nums.count(can)>n//3:
                res.append(can)
        return res
