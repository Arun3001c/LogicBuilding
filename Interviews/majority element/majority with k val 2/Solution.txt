class Solution(object):
    def majorityElement(self, nums):
        """
        :type nums: List[int]
        :rtype: int
        """
        count=0
        candidate=-1
        for num in nums:
            if candidate==num:
                count+=1
            elif count==0:
                candidate=num
                count=1
            else:
                count-=1
        freq=0
        for num in nums:
            if candidate==num:
                freq+=1
        if freq>len(nums)/2:
            return candidate
