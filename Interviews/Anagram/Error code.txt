
def is_anagram(s,t):
    s_dict={}
    t_dict={}
    for char in s:
        s_dict[char]=s_dict.get(char,0) + 1
    for char in t:
        t_dict[char]=t_dict.get(char,0)+1
    for key in s_dict.keys():
        if s_dict[key] != t_dict[key]:
            return False
    return True
    
print(is_anagram(s = "anagram",t = "nagaram"))
