def max_digit_sum_number(n):
    best = n
    best_sum = sum(map(int, str(n)))

    s = str(n)

    for i in range(len(s)):
        if s[i] == '0':
            continue

        temp = list(s)
        temp[i] = str(int(temp[i]) - 1)

        for j in range(i + 1, len(s)):
            temp[j] = '9'

        num = int("".join(temp))

        if num <= n:
            digit_sum = sum(map(int, str(num)))

            if digit_sum > best_sum:
                best_sum = digit_sum
                best = num

    return best


n = int(input())
print(max_digit_sum_number(n))
