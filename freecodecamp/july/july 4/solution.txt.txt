def kaprekar(n):
    count = 0

    # If already 6174, no steps needed
    if n == 6174:
        return 0

    while n != 6174:
        # Convert number to 4-digit string (important for leading zeros)
        num_str = f"{n:04d}"

        # Create ascending and descending numbers
        asc = int("".join(sorted(num_str)))
        desc = int("".join(sorted(num_str, reverse=True)))

        # Kaprekar step
        n = desc - asc

        count += 1

        # Safety check (handles edge case like 0000)
        if n == 0:
            return count

    return count


print(kaprekar(1234))
