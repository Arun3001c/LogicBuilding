def is_pronic(n):
    values=[]
    if n == 0:
        return True
    def factors(n):
        for i in range(1,n+1):
            if n%i==0:
                values.append(i)
    factors(n)

    for i in range(1,len(values)):
        if values[i]==values[i-1] +1 and values[i] * values[i-1] == n:
            return True
        

    return False
