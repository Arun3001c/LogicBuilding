def piggy_bank(coins):
    values={
        "pennies":0.01,
        "nickels":0.05,
        "dimes":0.10,
        "quarters":0.25
        }
    total=0
    for coin in coins:
        if coin in values:
            total+= values[coin] * coins[coin]
