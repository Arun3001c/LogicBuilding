import math
def get_max_profit(arr, budget):
    n = len(arr)

    max_profit = 0

    for i in range(n - 1):

        buy_price = arr[i]

        sell_price = arr[i + 1]

        for j in range(i + 1, n):

            if arr[j] > sell_price:
                sell_price = arr[j]

        shares = budget//buy_price

        profit = shares*sell_price - shares*buy_price 

        if profit > max_profit:
            max_profit = profit
        rounded_down = math.floor(max_profit * 100) / 100
        result = "{:.2f}".format(rounded_down)
    return result

# return ...
print(get_max_profit([5, 6], 50))
print(get_max_profit([8, 2, 5, 10], 20))
print(get_max_profit([4, 5, 3, 6], 20))
print(get_max_profit([54.40, 51.22, 53.99, 50.28, 53.01, 52.84], 200))
print(get_max_profit([15.38, 15.01, 14.99, 14.62, 14.28], 80))
