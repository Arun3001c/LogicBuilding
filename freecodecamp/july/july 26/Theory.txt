# Forward distance means: starting from the current letter (start),
# how many steps do we move forward to reach the target letter (end)?
# Example: a -> d = 3 steps, so we calculate end - start.
# % 26 is used because the alphabet is a circle. It wraps around after z.
forward = (end - start) % 26


# Backward distance means: starting from the current letter (start),
# how many steps do we move backward to reach the target letter (end)?
# Example: d -> a = 3 steps, so we calculate start - end.
# % 26 handles cases where we cross the beginning of the alphabet.
backward = (start - end) % 26


# Since the alphabet is circular, the shortest path can be either forward
# or backward, so we choose the smaller distance.
distance = min(forward, backward)
