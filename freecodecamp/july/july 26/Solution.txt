def letter_distance(str1, str2):
    alphabets = "abcdefghijklmnopqrstuvwxyz"
    count = 0

    for s, e in zip(str1, str2):
        start = alphabets.index(s)
        end = alphabets.index(e)

        forward = (end - start) % 26
        backward = (start - end) % 26

        count += min(forward, backward)

    return count
