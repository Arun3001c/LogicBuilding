def blend_words(word1, word2):
    len1=len(word1)//2
    len2=len(word2)//2
    new_word=[]
    for i in range (0,len1):
        new_word.append(word1[i])
    for i in range ( len2,len(word2)):
        new_word.append(word2[i])
        
    
    return "".join(new_word)

print( blend_words("turtle", "toucan"))

print ( blend_words("hyena", "iguana") )

def blend_words(word1, word2):
    # First half of the first word (shorter half if odd)
    first_half = word1[:len(word1) // 2]

    # Second half of the second word (longer half if odd)
    second_half = word2[len(word2) // 2:]

    return first_half + second_half
