def five_dice(dice):
    values={}
    for i in dice:
        if not i in values:
            values[i]=1
        else:
            values[i]+=1
    if len(values)==1:
        return "five of a kind"
    elif len(values)==2:
        if 4 in values.values():
            return "four of a kind"
        else:
            return "full house"
    elif len(values)==3:
        if 2 in values.values():
            return "two pair"
        else:
            return "three of a kind"
    elif len(values)==4:
        return "pair"
    else:
        unique = sorted(set(dice))
        if unique == [1,2,3,4,5] or unique == [2,3,4,5,6]:
            return "large straight"
        if ([1,2,3,4] == unique[:4] or [2,3,4,5] == unique[:4] or [3,4,5,6] == unique[:4] or (len(unique) == 5 and ( unique[1:] == [2,3,4,5] or unique[1:] == [3,4,5,6]))):
            return "small straight"
    return "no pair"
