def bucket_fill(grid, pos, new_value):
    rows = len(grid)
    cols = len(grid[0])

    old_value = grid[pos[0]][pos[1]]

    if old_value == new_value:
        return grid

    def dfs(r, c):
        if r < 0 or r >= rows or c < 0 or c >= cols:
            return

        if grid[r][c] != old_value:
            return

        grid[r][c] = new_value

        dfs(r - 1, c)  # up
        dfs(r + 1, c)  # down
        dfs(r, c - 1)  # left
        dfs(r, c + 1)  # right

    dfs(pos[0], pos[1])

    return grid
