def nth_fibonacci(n):
    if n==1:
        return 0
    if n==2:
        return 1
    first_number=0
    second_number=1
    result=0
    for i in range(3,n+1):
        result=first_number+second_number
        first_number=second_number
        second_number=result
    return result

print(nth_fibonacci(10),end="")
