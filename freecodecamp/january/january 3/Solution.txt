#a cleaner and more Pythonic version that keeps essentially the same algorithm:

def find_left_handed_seats(table):
    count = 0

    for row_index, row in enumerate(table):
        for seat_index, seat in enumerate(row):
            if seat != "U":
                continue

            if row_index == 0:  # Top row faces down
                if seat_index == len(row) - 1 or row[seat_index + 1] != "R":
                    count += 1
            else:  # Bottom row faces up
                if seat_index == 0 or row[seat_index - 1] != "R":
                    count += 1

    return count

#even more clearer version
def find_left_handed_seats(table):
    count = 0

    for row_index, row in enumerate(table):
        direction = 1 if row_index == 0 else -1

        for seat_index, seat in enumerate(row):
            if seat != "U":
                continue

            left = seat_index + direction

            if not (0 <= left < len(row)) or row[left] != "R":
                count += 1

    return count
