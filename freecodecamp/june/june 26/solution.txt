def triage_blood(bank, patients):
    def convert(arr):
        counts = {}
        for blood in arr:
            counts[blood] = counts.get(blood, 0) + 1
        return counts

    inventory = convert(bank)
    patient = convert(patients)

    served = 0

    # O patients
    use = min(patient.get("O", 0), inventory.get("O", 0))
    served += use
    patient["O"] = patient.get("O", 0) - use
    inventory["O"] = inventory.get("O", 0) - use

    # A patients
    use = min(patient.get("A", 0), inventory.get("A", 0))
    served += use
    patient["A"] = patient.get("A", 0) - use
    inventory["A"] = inventory.get("A", 0) - use
    
    use = min(patient.get("A", 0), inventory.get("O", 0))
    served += use
    patient["A"] -= use
    inventory["O"] -= use

    # B patients
    use = min(patient.get("B", 0), inventory.get("B", 0))
    served += use
    patient["B"] = patient.get("B", 0) - use
    inventory["B"] = inventory.get("B", 0) - use

    use = min(patient.get("B", 0), inventory.get("O", 0))
    served += use
    patient["B"] -= use
    inventory["O"] -= use
    
    # AB patients
    for donor in ["AB", "A", "B", "O"]:
        use = min(patient.get("AB", 0), inventory.get(donor, 0))
        served += use
        patient["AB"] = patient.get("AB", 0) - use
        inventory[donor] = inventory.get(donor, 0) - use

    return f"{served} of {len(patients)} patients served"
print(triage_blood(["O", "A", "B", "AB"], ["O", "A", "B", "AB"]))
