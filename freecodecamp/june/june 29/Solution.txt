#dictionary based logic
def get_mood(genre, bpm):
    moods = {
        "classical": [
            (60, 109, "focus"),
            (110, 180, "happy")
        ],
        "pop": [
            (60, 180, "happy")
        ],
        "rock": [
            (60, 129, "happy"),
            (130, 180, "hype")
        ],
        "electronic": [
            (60, 89, "focus"),
            (90, 134, "happy"),
            (135, 180, "hype")
        ]
    }

    for low, high, mood in moods.get(genre, []):
        if low <= bpm <= high:
            return mood
