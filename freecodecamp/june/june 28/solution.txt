def connect_three(board):
    rows = len(board)
    cols = len(board[0])

    for row in range(rows):
        for col in range(cols):

            # Skip empty cells
            if board[row][col] == "":
                continue

            player = board[row][col]

            # Check Horizontal (→)
            if col + 2 < cols:
                if board[row][col + 1] == player and board[row][col + 2] == player:
                    return [player, [row, col], [row, col + 1], [row, col + 2]]

            # Check Vertical (↓)
            if row + 2 < rows:
                if board[row + 1][col] == player and board[row + 2][col] == player:
                    return [player, [row, col], [row + 1, col], [row + 2, col]]

            # Check Diagonal (↘)
            if row + 2 < rows and col + 2 < cols:
                if board[row + 1][col + 1] == player and board[row + 2][col + 2] == player:
                    return [player, [row, col], [row + 1, col + 1], [row + 2, col + 2]]

            # Check Diagonal (↙)
            if row + 2 < rows and col - 2 >= 0:
                if board[row + 1][col - 1] == player and board[row + 2][col - 2] == player:
                    return [player, [row, col], [row + 1, col - 1], [row + 2, col - 2]]

    # No winner found
    return []


# Example
board = [
    ["", "", "", ""],
    ["", "", "", ""],
    ["", "Y", "", ""],
    ["Y", "R", "R", "R"]
]

print(connect_three(board))
