def is_valid_nonogram(clue, cells):
    index=0
    n=len(cells)
    for num in clue:
        count=0
        while index < n and cells[index]==0  :
            index+=1
        while index < n and cells[index]!=0:
            count+=1
            index+=1
        
        # print( num, count)
        if count!=num:
            return False
    while index<n:
        if cells[index]==1:
            return False
        index+=1
    return True
