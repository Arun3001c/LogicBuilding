def is_valid_number(n, base):
    n=n.upper()
    if base==2:
        for char in  n:
            if char not in "01":
                return False
    elif base==8:
        for char in n:
            if char not in "01234567":
                return False
    elif base==10:
        for char in n:
            if char not in "0123456789":
                return False
    elif base==16:
        for char in n:
            if char not in "0123456789ABCDEF":
                return False 
    elif base==32:
        character='0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ'
        for char in n:
            if not  char in character:
                return False
        
    return True
